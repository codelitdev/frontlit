## Usage

```tsx
import {
    createWidgetInstance,
    PageRenderer,
    getDefaultTheme,
} from "@frontlit/page-builder";

const layout = [createWidgetInstance("faq")];

<PageRenderer
    layout={layout}
    theme={getDefaultTheme()}
    pageData={{ pageType: "landing" }}
/>;
```
