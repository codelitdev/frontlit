<PageBuilderDemo demo="footer" />

## Usage

```tsx
import {
    createWidgetInstance,
    PageRenderer,
    getDefaultTheme,
} from "@frontlit/page-builder";

const layout = [createWidgetInstance("footer")];

<PageRenderer
    layout={layout}
    theme={getDefaultTheme()}
    pageData={{ pageType: "landing" }}
/>;
```
